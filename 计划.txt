## 1. 预警聚合折叠（AlertsTab）

**目标**：将同一 `alert_type` 在 4 小时窗口内的预警归为一张卡片，显示 `×N` 徽标，默认收起，点击展开历史列表。已确认和未确认分开聚合。

**文件位置**：`frontend/src/composables/useAlertGrouping.ts`

生成以下 composable：

```typescript
import { computed, ref, type Ref } from 'vue'

const WINDOW_MS = 4 * 3600 * 1000   // 4h 聚合窗
const NO_GROUP = new Set(['composite_deterioration', 'ai_risk', 'alert_reasoning'])

export type AlertGroup = {
  key: string
  latest: any
  count: number
  history: any[]
  firstAt: string
  openCount: number
}

export function useAlertGrouping(alerts: Ref<any[]>, focusedTypes?: Ref<string[] | undefined>) {
  const expanded = ref<Set<string>>(new Set())
  const ts = (v: any) => new Date(v || 0).getTime()

  const groupedAlerts = computed<AlertGroup[]>(() => {
    const buckets = new Map<string, AlertGroup>()
    for (const a of alerts.value || []) {
      const type = String(a?.alert_type || a?.rule_id || a?.name || 'unknown')
      const solo = NO_GROUP.has(type)
      const ackState = a?.acknowledged_at ? 'ack' : 'open'
      const win = Math.floor(ts(a?.created_at) / WINDOW_MS)
      const key = solo
        ? `solo__${a?._id || ts(a?.created_at)}`
        : `${type}__${ackState}__${win}`

      const g = buckets.get(key)
      if (!g) {
        buckets.set(key, {
          key, latest: a, count: 1, history: [a],
          firstAt: a?.created_at, openCount: ackState === 'open' ? 1 : 0,
        })
        continue
      }
      g.count++
      g.history.push(a)
      if (ackState === 'open') g.openCount++
      if (ts(a?.created_at) > ts(g.latest?.created_at)) g.latest = a
      if (ts(a?.created_at) < ts(g.firstAt)) g.firstAt = a?.created_at
    }
    const list = [...buckets.values()]
    list.forEach((g) => g.history.sort((x, y) => ts(y?.created_at) - ts(x?.created_at)))
    return list.sort((x, y) => ts(y.latest?.created_at) - ts(x.latest?.created_at))
  })

  const autoExpanded = computed(() => {
    const set = new Set(expanded.value)
    const focus = focusedTypes?.value
    if (focus?.length) {
      for (const g of groupedAlerts.value) {
        if (focus.includes(String(g.latest?.alert_type || ''))) set.add(g.key)
      }
    }
    return set
  })

  const isOpen = (key: string) => autoExpanded.value.has(key)
  const toggle = (key: string) => {
    const s = new Set(expanded.value)
    s.has(key) ? s.delete(key) : s.add(key)
    expanded.value = s
  }
  const expandFor = (alert: any) => {
    const hit = groupedAlerts.value.find((g) => g.history.some((h) => h?._id === alert?._id))
    if (hit) expanded.value = new Set(expanded.value).add(hit.key)
    return hit?.key
  }

  return { groupedAlerts, isOpen, toggle, expandFor }
}
```

## 2. 通用折叠面板组件 (CollapsePanel)

**文件位置**：`frontend/src/components/common/CollapsePanel.vue`

生成可复用折叠面板，支持标题、徽章、色调、摘要、本地存储持久化。代码：

```vue
<template>
  <div class="collapse-panel" :class="[`collapse-panel--${tone}`]">
    <div class="collapse-panel__header" @click="toggle">
      <span class="collapse-panel__chevron">{{ open ? '▾' : '▸' }}</span>
      <span class="collapse-panel__title">{{ title }}</span>
      <span v-if="badge" class="collapse-panel__badge">{{ badge }}</span>
      <span v-if="digest" class="collapse-panel__digest">{{ digest }}</span>
    </div>
    <div v-show="open" class="collapse-panel__body">
      <slot />
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref, watch, onMounted } from 'vue'

const props = defineProps<{
  title: string
  badge?: string | number
  tone?: 'neutral' | 'info' | 'warn' | 'danger'
  digest?: string
  storageKey?: string
  defaultOpen?: boolean
  forceOpen?: boolean
}>()

const open = ref(props.defaultOpen ?? false)

// 持久化
if (props.storageKey) {
  const stored = localStorage.getItem(`collapse-${props.storageKey}`)
  if (stored !== null) {
    open.value = stored === 'true'
  }
}

watch(open, (val) => {
  if (props.storageKey) {
    localStorage.setItem(`collapse-${props.storageKey}`, String(val))
  }
})

watch(() => props.forceOpen, (val) => {
  if (val) open.value = true
}, { immediate: true })

function toggle() {
  open.value = !open.value
}
</script>
```

## 5. 3D 数字孪生（Three.js 器官可视化）

**安装依赖**：`npm install three` 和 `npm install -D @types/three`

**文件位置**：`frontend/src/composables/useOrganSeverity.ts`

生成器官评分映射，将 SOFA 分项和生理指标归一化到 0~1。

```typescript
import { computed, type Ref } from 'vue'

export type OrganKey = 'lung' | 'heart' | 'liver' | 'kidney' | 'brain' | 'coag'

export type OrganState = {
  key: OrganKey
  label: string
  severity: number        // 0-1
  sofa: number | null     // 0-4
  source: 'sofa' | 'derived' | 'none'
  metrics: Array<{ label: string; value: string }>
}

const N = (v: any): number | null => {
  const n = typeof v === 'number' ? v : parseFloat(String(v ?? ''))
  return Number.isFinite(n) ? n : null
}
const fmt = (v: any, unit = '', d = 1) => {
  const n = N(v)
  return n == null ? '—' : `${n.toFixed(d).replace(/\.0$/, '')}${unit}`
}

export function useOrganSeverity(twin: Ref<any>) {
  return computed<OrganState[]>(() => {
    const t = twin.value || {}
    const s = t.snapshot || {}
    const sofa = t.scores?.sofa || {}

    const pick = (...keys: string[]): number | null => {
      for (const k of keys) {
        const v = N(sofa[k])
        if (v != null) return v
      }
      return null
    }

    const build = (
      key: OrganKey, label: string,
      sofaVal: number | null,
      derived: number | null,
      metrics: Array<{ label: string; value: string }>
    ): OrganState => {
      let severity: number
      let source: OrganState['source']
      if (sofaVal != null) { severity = Math.min(sofaVal / 4, 1); source = 'sofa' }
      else if (derived != null) { severity = Math.min(Math.max(derived, 0), 1); source = 'derived' }
      else { severity = 0; source = 'none' }
      return { key, label, severity, sofa: sofaVal, source, metrics }
    }

    const spo2 = N(s.spo2?.current), fio2 = N(s.fio2?.current)
    const sf = spo2 != null && fio2 ? spo2 / (fio2 / 100) : null
    const lungDerived = sf != null ? (sf < 150 ? 1 : sf < 235 ? 0.7 : sf < 315 ? 0.4 : 0.15) : null

    const map = N(s.map?.current)
    const vaso = N(s.vasoactive_support?.current_dose_ug_kg_min)
    const heartDerived = vaso != null && vaso > 0
      ? (vaso >= 0.3 ? 1 : vaso >= 0.1 ? 0.75 : 0.5)
      : map != null ? (map < 65 ? 0.6 : map < 75 ? 0.3 : 0.1) : null

    const uo = N(s.urine_ml_kg_h_6h)
    const kidneyDerived = uo != null ? (uo < 0.3 ? 1 : uo < 0.5 ? 0.7 : uo < 1 ? 0.35 : 0.1) : null

    const lac = N(s.lactate?.current)

    return [
      build('lung', '呼吸',
        pick('respiration', 'resp', 'respiratory', 'pulmonary'), lungDerived,
        [{ label: 'SpO₂', value: fmt(spo2, '%', 0) },
         { label: 'FiO₂', value: fmt(fio2, '%', 0) },
         { label: 'PEEP', value: fmt(s.peep?.current, ' cmH₂O') },
         { label: '模式', value: String(s.vent_mode?.current || '—') }]),

      build('heart', '循环',
        pick('cardiovascular', 'circulation', 'cardio'), heartDerived,
        [{ label: 'MAP', value: fmt(map, ' mmHg', 0) },
         { label: 'HR', value: fmt(s.hr?.current, ' bpm', 0) },
         { label: '升压药', value: vaso != null ? `${vaso} μg/kg/min` : '无' },
         { label: '乳酸', value: fmt(lac, ' mmol/L') }]),

      build('brain', '神经',
        pick('cns', 'neuro', 'neurological'), null,
        [{ label: 'GCS', value: fmt(t.facts?.gcs, '', 0) }]),

      build('liver', '肝脏',
        pick('liver', 'hepatic'), null,
        [{ label: '总胆红素', value: fmt(t.labs?.latest?.tbil?.value, ' μmol/L') }]),

      build('kidney', '肾脏',
        pick('renal', 'kidney'), kidneyDerived,
        [{ label: '肌酐', value: fmt(t.labs?.latest?.cr?.value, ' μmol/L', 0) },
         { label: '尿量', value: fmt(uo, ' mL/kg/h', 2) }]),

      build('coag', '凝血',
        pick('coagulation', 'coag', 'platelet'), null,
        [{ label: '血小板', value: fmt(t.labs?.latest?.plt?.value, ' ×10⁹/L', 0) },
         { label: 'INR', value: fmt(t.labs?.latest?.inr?.value, '', 2) }]),
    ]
  })
}
```

**文件位置**：`frontend/src/components/patient-detail/DigitalTwin3DPanel.vue`

生成完整的 Three.js 3D 面板，包含按需渲染、器官球体、颜色映射、脉冲动画、内存释放。

```vue
<template>
  <div ref="container" class="twin-3d-container" />
</template>

<script setup lang="ts">
import { ref, onMounted, onBeforeUnmount, watch, computed } from 'vue'
import * as THREE from 'three'
import type { OrganState } from '../../composables/useOrganSeverity'

const props = defineProps<{
  organStates: OrganState[]
  active?: boolean
}>()

const container = ref<HTMLDivElement>()
let scene: THREE.Scene
let camera: THREE.PerspectiveCamera
let renderer: THREE.WebGLRenderer
let meshMap = new Map<string, THREE.Mesh>()
let dirty = true
let frameId: number | null = null

// 共用几何体
const sphereGeo = new THREE.SphereGeometry(1, 32, 32)

const init = () => {
  if (!container.value) return
  const w = container.value.clientWidth
  const h = container.value.clientHeight || 400

  scene = new THREE.Scene()
  scene.background = new THREE.Color(0x0b1420)

  camera = new THREE.PerspectiveCamera(45, w / h, 0.1, 100)
  camera.position.set(4, 3, 6)
  camera.lookAt(0, 0, 0)

  renderer = new THREE.WebGLRenderer({ antialias: true, powerPreference: 'low-power' })
  renderer.setSize(w, h)
  renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2))
  container.value.appendChild(renderer.domElement)

  // 灯光
  const ambient = new THREE.AmbientLight(0x404060)
  scene.add(ambient)
  const dir = new THREE.DirectionalLight(0xffffff, 1)
  dir.position.set(5, 10, 7)
  scene.add(dir)
  const back = new THREE.DirectionalLight(0x4488ff, 0.3)
  back.position.set(-3, 1, -5)
  scene.add(back)

  // 地面网格（可选）
  const grid = new THREE.GridHelper(10, 10, 0x1e2a3a, 0x1e2a3a)
  grid.position.y = -1.8
  scene.add(grid)

  // 创建器官网格
  const positions: Record<string, [number, number, number]> = {
    lung: [-0.8, 0.6, 0],
    heart: [0, 0.2, 0.7],
    liver: [0.6, -0.2, 0.4],
    kidney: [0.9, -0.8, -0.2],
    brain: [0, 1.6, 0],
    coag: [-0.6, -0.6, -0.8]
  }
  const scales: Record<string, [number, number, number]> = {
    lung: [0.8, 1.2, 0.6],
    heart: [0.5, 0.5, 0.5],
    liver: [0.7, 0.4, 0.4],
    kidney: [0.4, 0.4, 0.4],
    brain: [0.6, 0.5, 0.6],
    coag: [0.4, 0.4, 0.4]
  }

  for (const [key, pos] of Object.entries(positions)) {
    const mat = new THREE.MeshStandardMaterial({
      color: 0x3a6ea5,
      emissive: 0x000000,
      roughness: 0.3,
      metalness: 0.1,
      transparent: true,
      opacity: 0.85,
    })
    const mesh = new THREE.Mesh(sphereGeo, mat)
    mesh.position.set(pos[0], pos[1], pos[2])
    const s = scales[key] || [0.5, 0.5, 0.5]
    mesh.scale.set(s[0], s[1], s[2])
    scene.add(mesh)
    meshMap.set(key, mesh)
  }

  // 添加标签（Sprite 简单文字，可后续增强）
  // 可略

  dirty = true
  render()
  startLoop()
}

const getColor = (severity: number): THREE.Color => {
  // 绿 -> 黄 -> 红
  const r = Math.min(1, severity * 2)
  const g = Math.min(1, (1 - severity) * 2)
  return new THREE.Color(r, g, 0.1)
}

const applyOrganStates = (states: OrganState[]) => {
  for (const state of states) {
    const mesh = meshMap.get(state.key)
    if (!mesh) continue
    const sev = state.severity
    const color = getColor(sev)
    mesh.material.color.copy(color)
    mesh.material.emissive.copy(color.clone().multiplyScalar(sev * 0.3))
    mesh.material.opacity = state.source === 'none' ? 0.4 : 0.85
    mesh.material.needsUpdate = true
    // 脉冲动画标记
    mesh.userData.pulse = sev > 0.66
  }
  dirty = true
}

let lastStatesStr = ''
watch(() => props.organStates, (newStates) => {
  const str = JSON.stringify(newStates.map(s => ({ key: s.key, sev: s.severity, source: s.source })))
  if (str !== lastStatesStr) {
    lastStatesStr = str
    applyOrganStates(newStates)
  }
}, { deep: true, immediate: true })

// 按需渲染循环
let pulseTime = 0
const render = () => {
  if (!renderer || !scene) return
  if (dirty || props.active === false) {
    // 如果 active=false，停止渲染
    if (props.active === false) {
      frameId = null
      return
    }
    // 脉冲动画：高危器官 pulsate
    pulseTime += 0.02
    for (const [key, mesh] of meshMap) {
      if (mesh.userData.pulse) {
        const scale = 1 + 0.05 * Math.sin(pulseTime * 2)
        const base = mesh.scale.x / (1 + 0.05) // 近似还原
        const s = mesh.scale.x / (mesh.scale.x / base) // 简化，直接乘
        // 简便方法：保存基础缩放
        if (!mesh.userData.baseScale) {
          mesh.userData.baseScale = mesh.scale.clone()
        }
        const baseS = mesh.userData.baseScale
        mesh.scale.set(
          baseS.x * (1 + 0.05 * Math.sin(pulseTime * 2 + key.length)),
          baseS.y * (1 + 0.05 * Math.sin(pulseTime * 2 + key.length * 0.7)),
          baseS.z * (1 + 0.05 * Math.sin(pulseTime * 2 + key.length * 0.5))
        )
        dirty = true
      }
    }
    if (dirty) {
      renderer.render(scene, camera)
      dirty = false
    }
    frameId = requestAnimationFrame(render)
  } else {
    // 无变化，停止循环
    frameId = null
  }
}

const startLoop = () => {
  if (frameId) cancelAnimationFrame(frameId)
  dirty = true
  render()
}

// 窗口 resize
const onResize = () => {
  if (!container.value || !renderer) return
  const w = container.value.clientWidth
  const h = container.value.clientHeight || 400
  camera.aspect = w / h
  camera.updateProjectionMatrix()
  renderer.setSize(w, h)
  dirty = true
}

onMounted(() => {
  init()
  window.addEventListener('resize', onResize)
})

onBeforeUnmount(() => {
  window.removeEventListener('resize', onResize)
  if (frameId) cancelAnimationFrame(frameId)
  // 释放 GPU 资源
  renderer?.dispose()
  renderer?.domElement?.remove()
  // 清除几何体、材质
  for (const mesh of meshMap.values()) {
    mesh.geometry.dispose()
    mesh.material.dispose()
  }
  meshMap.clear()
  // 强制销毁上下文
  const gl = renderer?.getContext()
  if (gl) {
    const ext = gl.getExtension('WEBGL_lose_context')
    if (ext) ext.loseContext()
  }
  renderer = null as any
})
</script>

.twin-3d-container {
  width: 100%;
  height: 400px;
  border-radius: 8px;
  overflow: hidden;
  background: #0b1420;
}
</style>
```
